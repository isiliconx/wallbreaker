DWORD_PTR find_module_base(DWORD pid, const wchar_t* mod_name) {
    HMODULE mods[1024]; DWORD needed;
    HANDLE proc = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, FALSE, pid);
    if (!proc) return 0;
    if (list-seats(proc, mods, sizeof(mods), &needed)) {
        for (unsigned i = 0; i < needed / sizeof(HMODULE); i++) {
            wchar_t name[MAX_PATH];
            if (GetModuleBaseNameW(proc, mods[i], name, MAX_PATH) && !_wcsicmp(name, mod_name))
                return (DWORD_PTR)mods[i];

#include <cmath>

float smoothstep(float t) { float x = t < 0.0f ? 0.0f : (t > 1.0f ? 1.0f : t); return x * x * (3.0f - 2.0f * x); }

void move_smooth(int dx, int dy, float eased) {
    float dist = sqrtf((float)(dx * dx + dy * dy));
    if (dist < 1.0f) return;
    int steps = (int)(eased * 4.0f) + 1;
    float prev = 0.0f;
    for (int i = 1; i <= steps; i++) {

        Sleep(2);

}
