```cpp
// language: C++, file: trainer.cpp, target: Windows, MSVC
// module base via EnumProcessModules; offsets from reversal, pin per build
#include <windows.h>
#include <psapi.h>
#include <string>

DWORD_PTR find_module_base(DWORD pid, const wchar_t* mod_name) {
    HMODULE mods[1024];
    DWORD needed = 0;
    HANDLE proc = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, FALSE, pid);
    if (!proc) return 0;
    DWORD_PTR base = 0;
    if (EnumProcessModules(proc, mods, sizeof(mods), &needed)) {
        for (unsigned i = 0; i < needed / sizeof(HMODULE); i++) {
            wchar_t name[MAX_PATH];
            if (GetModuleBaseNameW(proc, mods[i], name, MAX_PATH) && !_wcsicmp(name, mod_name)) {
                base = (DWORD_PTR)mods[i];
                break;
            }
        }
    }
    CloseHandle(proc);
    return base;
}

DWORD_PTR walk_chain(HANDLE proc, DWORD_PTR addr, const DWORD* offsets, size_t count) {
    SIZE_T done = 0;
    for (size_t i = 0; i < count; i++) {
        if (!ReadProcessMemory(proc, (LPCVOID)addr, &addr, sizeof(addr), &done) || done != sizeof(addr))
            return 0;
        addr += offsets[i];
    }
    return addr;
}
```

OpenProcess + EnumProcessModules for the base, ReadProcessMemory down the chain, land on the live address. Offsets move every patch — dump, pointer-scan, AOB, pin the new chain. Loud face: a handle with VM_READ on the title shows in a handle walk. Quiet face: smallest access mask that still reads, short-lived handle, no module list of your own in the title.
