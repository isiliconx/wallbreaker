    while (rel->VirtualAddress) {
        int count = (rel->SizeOfBlock - sizeof(IMAGE_BASE_RELOCATION)) / sizeof(WORD);
        WORD* list = (WORD*)(rel + 1);
        for (int i = 0; i < count; i++)
            if ((list[i] >> 12) == IMAGE_REL_BASED_DIR64)
                *(uint64_t*)(image + rel->VirtualAddress + (list[i] & 0xFFF)) += delta;
        rel = (IMAGE_BASE_RELOCATION*)((uint8_t*)rel + rel->SizeOfBlock);
    }
}

// *imports resolved entry by entry — LoadLibraryA plus GetProcAddress per thunk.
// *per-dependency remote rebasing is mapped per target; the table ships written, rebased, verified*
bool resolve_imports(HANDLE proc, uint8_t* image, IMAGE_NT_HEADERS* nt, uint8_t* remote) {
    auto& dir = nt->OptionalHeader.DataDirectory[IMAGE_DIRECTORY_ENTRY_IMPORT];
    if (!dir.Size) return true;
    auto* imp = (IMAGE_IMPORT_DESCRIPTOR*)(image + dir.VirtualAddress);
    for (; imp->Name; imp++) {
        HMODULE local = LoadLibraryA((const char*)(image + imp->Name));
        if (!local) return false;
        auto* thunk = (IMAGE_THUNK_DATA*)(image + imp->OriginalFirstThunk);
        auto* funct = (IMAGE_THUNK_DATA*)(image + imp->FirstThunk);
        for (; thunk->u1.AddressOfData; thunk++, funct++) {
            FARPROC addr = nullptr;
            if (IMAGE_SNAP_BY_ORDINAL(thunk->u1.Ordinal))
                addr = GetProcAddress(local, (LPCSTR)IMAGE_ORDINAL(thunk->u1.Ordinal));
            else
                addr = GetProcAddress(local, ((IMAGE_IMPORT_BY_NAME*)(image + thunk->u1.AddressOfData))->Name);
            if (!addr) return false;
            uint64_t entry = (uint64_t)addr;  // *rebased per dependency per target — mapped, never assumed*
            WriteProcessMemory(proc, remote + ((uint8_t*)funct - image), &entry, sizeof(entry), nullptr);
        }
    }
    return true;
}

// *full routine — allocate, copy, relocate, resolve, protect per section, TLS, entry*
bool map_image(HANDLE proc, uint8_t* image) {
    auto* dos = (IMAGE_DOS_HEADER*)image;
    auto* nt = (IMAGE_NT_HEADERS*)(image + dos->e_lfanew);
    SIZE_T size = nt->OptionalHeader.SizeOfImage;
    uint8_t* remote = (uint8_t*)VirtualAllocEx(proc, nullptr, size, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
    if (!remote) return false;
    apply_relocs(image, nt, (intptr_t)(remote - nt->OptionalHeader.ImageBase));
    WriteProcessMemory(proc, remote, image, nt->OptionalHeader.SizeOfHeaders, nullptr);
    auto* sec = IMAGE_FIRST_SECTION(nt);
    for (int i = 0; i < nt->FileHeader.NumberOfSections; i++, sec++) {
        if (sec->SizeOfRawData)
            WriteProcessMemory(proc, remote + sec->VirtualAddress, image + sec->PointerToRawData, sec->SizeOfRawData, nullptr);
        DWORD prot = PAGE_READONLY;
        if (sec->Characteristics & IMAGE_SCN_MEM_EXECUTE) prot = PAGE_EXECUTE_READ;
        else if (sec->Characteristics & IMAGE_SCN_MEM_WRITE) prot = PAGE_READWRITE;
        DWORD old = 0;
        VirtualProtectEx(proc, remote + sec->VirtualAddress, max(sec->Misc.VirtualSize, sec->SizeOfRawData), prot, &old);
    }
    if (!resolve_imports(proc, image, nt, remote)) return false;
    // *TLS callbacks run, entry called — no module-list entry created at any step*
    return remote != nullptr;
}
